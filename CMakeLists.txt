cmake_minimum_required(VERSION 3.20)
project(discord-leveler VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network DBus)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PULSE REQUIRED IMPORTED_TARGET libpulse-simple)
add_executable(discord-leveler src/main.cpp src/app.cpp src/app.h src/audio.cpp)
target_link_libraries(discord-leveler PRIVATE Qt6::Widgets Qt6::Network Qt6::DBus PkgConfig::PULSE)
target_compile_options(discord-leveler PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS discord-leveler RUNTIME DESTINATION bin)
enable_testing()
add_executable(dsp-test tests/dsp_test.cpp)
add_test(NAME dsp COMMAND dsp-test)

add_executable(routing-test tests/routing_test.cpp)
target_link_libraries(routing-test PRIVATE Qt6::Core)
add_test(NAME routing-isolation COMMAND routing-test)
